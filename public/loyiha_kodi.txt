export type TaskCategory = 'vazifa' | 'uchrashuv';
export type TaskPriority = 'low' | 'medium' | 'high';
export type TaskStatus = 'pending' | 'completed' | 'postponed' | 'cancelled';

export interface Task {
  id: string;
  title: string;
  category: TaskCategory;
  date: string; // YYYY-MM-DD
  time?: string; // HH:mm
  reminderTime?: string;
  status: TaskStatus;
  completedAt?: string;
  priority: TaskPriority;
  notes?: string;
  source: 'bot' | 'miniapp';
  postponedUntil?: string;
  isArchived?: boolean;
}

export type HabitFrequency = 'daily' | 'weekly_target' | 'specific_days';
export type HabitType = 'general' | 'supplement_health';

export interface Habit {
  id: string;
  title: string;
  category: 'odat';
  habitType?: HabitType;
  dosage?: string; // e.g. "5 gram", "1 stakan", "2 kapsula"
  timeSlot?: string; // e.g. "Ertalab", "Mashg'ulotdan oldin", "Kechqurun"
  frequency: HabitFrequency;
  targetDaysPerWeek?: number; // e.g. 3 for "Haftada 3 marta"
  daysOfWeek: number[]; // 1=Mon (Du), 2=Tue (Se), 3=Wed (Ch), 4=Thu (Pa), 5=Fri (Ju), 6=Sat (Sh), 7=Sun (Ya)
  hasQuantity: boolean;
  targetQuantity?: number;
  unit?: string; // e.g., 'marta', 'sahifa', 'daqiqa', 'litr', 'gram'
  streak: number;
  bestStreak: number;
  currentQuantityToday?: number;
  history: Record<string, { completed: boolean; quantity?: number; timestamp?: string }>;
  linkedGoalId?: string;
  notes?: string;
  isArchived?: boolean;
}

export type ExpenseCategory =
  | 'oziq_ovqat'
  | 'transport'

  | 'sport'
  | 'talim'

  date: string; // YYYY-MM-DD

  notes?: string;
  source: 'bot' | 'miniapp';

    const today = formatDateYMD(new Date());
    const timeStr = getTimeString();
    setGoals(prev =>
      prev.map(g => {
        if (g.id === goalId) {
          const isDecrease = g.goalType === 'decrease';
          const newVal = isDecrease
            ? Math.max(0, g.currentValue - amount)
            : Math.max(0, g.currentValue + amount);

          const achieved = isDecrease ? newVal <= g.targetValue : newVal >= g.targetValue;
          const remaining = isDecrease
            ? Math.max(0, newVal - g.targetValue)
            : Math.max(0, g.targetValue - newVal);

          const newLog: GoalProgressLog = {
            id: `log-${Date.now()}`,
            date: today,
            time: timeStr,
            amount,
            note: note || undefined,
            resultingCurrent: newVal,
            resultingRemaining: remaining,
          };

          return {
            ...g,
            currentValue: newVal,
            status: achieved ? 'achieved' : 'active',

      })
    );

          if (!targetLog) return g;

          const isDecrease = g.goalType === 'decrease';
          const revertedVal = isDecrease
            ? g.currentValue + targetLog.amount
            : Math.max(0, g.currentValue - targetLog.amount);

          const achieved = isDecrease ? revertedVal <= g.targetValue : revertedVal >= g.targetValue;

          return {
            ...g,
            currentValue: revertedVal,
            status: achieved ? 'achieved' : 'active',
            progressLogs: g.progressLogs.filter(l => l.id !== logId),

import { ParsedItem, ExpenseCategory, Currency } from '../types';

/**
 * Formats a Date object to YYYY-MM-DD using local time
 */
export function formatDateYMD(d: Date): string {
  const year = d.getFullYear();
  const month = String(d.getMonth() + 1).padStart(2, '0');
  const day = String(d.getDate()).padStart(2, '0');
  return `${year}-${month}-${day}`;
}

/**
 * Format Uzbek human friendly date label
 */
export function formatUzbekDateLabel(dateStr: string, timeStr?: string | null): string {
  if (!dateStr) return '';
  const today = formatDateYMD(new Date());
  
  const tomorrowDate = new Date();
  tomorrowDate.setDate(tomorrowDate.getDate() + 1);
  const tomorrow = formatDateYMD(tomorrowDate);

  const indinDate = new Date();
  indinDate.setDate(indinDate.getDate() + 2);
  const indin = formatDateYMD(indinDate);

  let label = dateStr;
  if (dateStr === today) {
    label = 'Bugun';
  } else if (dateStr === tomorrow) {
    label = 'Ertaga';
  } else if (dateStr === indin) {
    label = 'Indin';
  } else {
    const [y, m, d] = dateStr.split('-');
    const months = [
      'yanvar', 'fevral', 'mart', 'aprel', 'may', 'iyun',
      'iyul', 'avgust', 'sentabr', 'oktabr', 'noyabr', 'dekabr'
    ];
    const monthName = months[parseInt(m, 10) - 1] || m;
    label = `${parseInt(d, 10)}-${monthName}`;
  }

  if (timeStr) {
    return `${label}, soat ${timeStr}`;
  }
  return label;
}

/**
 * Format amount in currency: either UZS (e.g. 45 000 so'm) or USD (e.g. 1 400 $)
 */
export function formatMoney(amount: number, currency: Currency = 'UZS'): string {
  if (currency === 'USD') {
    return `${new Intl.NumberFormat('en-US').format(amount)} $`;
  }
  return `${new Intl.NumberFormat('uz-UZ').format(amount)} so'm`;
}

/**
 * Format amount in Uzbek so'm (e.g. 45 000 so'm)
 */
export function formatSom(amount: number): string {
  return formatMoney(amount, 'UZS');
}

/**
 * Local Rule-based NLP Parser for Uzbek commands and natural language input.
 * Supports:
 * - prefixes: "vazifa:", "odat:", "maqsad:", "uchrashuv:", "harajat:", "xarajat:", "qabul:"
 * - expenses: "harajat: 45000 tushlik", "sarf: 120000 taksi", "50000 so'm kitob"
 * - health & supplements: "5g kreatin", "l-arginin", "ertalab asalli suv ichish", "har kuni 5 gram kreatin"
 * - dates: "bugun", "ertaga", "indin", day names, "YYYY-MM-DD", "DD.MM.YYYY"
 * - times: "soat 15:30", "15:30 da", "18:45", "10 daqiqadan keyin", "2 soatdan keyin"
 * - quantities: "20 sahifa", "30 daqiqa", "2 litr", "5 km", "5 gram"
 */
export function parseUzbekTextLocal(rawText: string, now: Date = new Date()): ParsedItem {
  const text = rawText.trim();
  const lower = text.toLowerCase();

  let category: ParsedItem['category'] = 'noaniq';
  let targetDate: string | null = null;
  let targetTime: string | null = null;
  let quantity: number | null = null;
  let unit: string | null = null;
  let amount: number | null = null;
  let currency: Currency = 'UZS';
  let expenseCategory: ExpenseCategory | null = null;
  let dosage: string | null = null;
  let timeSlot: string | null = null;
  let daysOfWeek: (number | string)[] | null = null;
  let confidence = 0.7;

  let cleanText = text;

  // 1. Detect Category Prefix or Domain Keywords
  if (lower.startsWith('harajat:') || lower.startsWith('xarajat:') || lower.startsWith('sarf:') || lower.startsWith('to\'lov:')) {
    category = 'harajat';
    cleanText = text.replace(/^(harajat|xarajat|sarf|to'lov):\s*/i, '');
    confidence += 0.25;
  } else if (lower.startsWith('qabul:') || lower.startsWith('ichish:') || lower.startsWith('dori:')) {
    category = 'qabul';
    cleanText = text.replace(/^(qabul|ichish|dori):\s*/i, '');
    confidence += 0.25;
  } else if (lower.startsWith('vazifa:') || lower.startsWith('topshiriq:') || lower.startsWith('ish:')) {
    category = 'vazifa';
    cleanText = text.replace(/^(vazifa|topshiriq|ish):\s*/i, '');
    confidence += 0.2;
  } else if (lower.startsWith('odat:') || lower.startsWith('odatlanish:')) {
    category = 'odat';
    cleanText = text.replace(/^(odat|odatlanish):\s*/i, '');
    confidence += 0.2;
  } else if (lower.startsWith('maqsad:')) {
    category = 'maqsad';
    cleanText = text.replace(/^maqsad:\s*/i, '');
    confidence += 0.2;
  } else if (lower.startsWith('uchrashuv:') || lower.startsWith('meet:')) {
    category = 'uchrashuv';
    cleanText = text.replace(/^(uchrashuv|meet):\s*/i, '');
    confidence += 0.2;
  }

  // 2. Health & Supplements detection (Kreatin, L-Arginin, Asalli suv, vitaminlar)
  const isCreatine = /(creatin|kreatin)/i.test(lower);
  const isArginine = /(l\s*[-_]?\s*arginin|arginin)/i.test(lower);
  const isHoneyWater = /(asalli\s+suv|asal\s+suv|iliq\s+suv)/i.test(lower);
  const isSupplement = isCreatine || isArginine || isHoneyWater || /(vitamin|omega|protein|dori|kapsula)/i.test(lower);

  if (category === 'noaniq' && isSupplement) {
    category = 'qabul';
    confidence += 0.2;
  }

  if (isSupplement || category === 'qabul') {
    // Extract dosage
    const dosageMatch = cleanText.match(/(\d+(?:\.\d+)?)\s*(g|gr|gram|gramm|kapsula|stakan|qoshiq|dona)/i);
    if (dosageMatch) {
      dosage = `${dosageMatch[1]} ${dosageMatch[2].toLowerCase()}`;
      quantity = parseFloat(dosageMatch[1]);
      unit = dosageMatch[2].toLowerCase();
    } else if (isCreatine) {
      dosage = '5 gram';
      quantity = 5;
      unit = 'gram';
    } else if (isArginine) {
      dosage = '3 gram (yoki 1 porsiya)';
      quantity = 3;
      unit = 'gram';
    } else if (isHoneyWater) {
      dosage = '1 stakan iliq suv + 1 qoshiq asal';
      quantity = 1;
      unit = 'stakan';
    }

    // Extract time slot
    if (/nahorga|och\s+qoringa|ertalab/i.test(lower)) {
      timeSlot = 'Ertalab nahorga (Och qoringa)';
    } else if (/mashg'ulotdan\s+oldin|mashqdan\s+oldin|trenirovkadan\s+oldin/i.test(lower)) {
      timeSlot = 'Mashg\'ulotdan 30 daqiqa oldin';
    } else if (/mashg'ulotdan\s+keyin|trenirovkadan\s+keyin/i.test(lower)) {
      timeSlot = 'Mashg\'ulotdan so\'ng darhol';
    } else if (/kechqurun|yotishdan\s+oldin/i.test(lower)) {
      timeSlot = 'Kechqurun yotishdan oldin';
    } else {
      if (isHoneyWater) timeSlot = 'Ertalab nahorga (Och qoringa)';
      else if (isArginine) timeSlot = 'Mashg\'ulotdan oldin / Ertalab';
      else if (isCreatine) timeSlot = 'Kun davomida / Mashg\'ulotdan so\'ng';
      else timeSlot = 'Kunlik qabul';
    }
  }

  // 3. Expense (Harajat) detection and extraction (UZS and USD e.g. 1400$, iphone 1400 dollar)
  const isDollar = /\$|\bdollar\b|\busd\b/i.test(lower);
  const somMatch = cleanText.match(/(\d[\d\s.,]*)\s*(?:ming\s*)?(?:so'?m|som|uzs|\$|dollar|usd)?/i);
  if (category === 'harajat' || isDollar || (somMatch && /(so'?m|som|ming|sarfladim|to'ladim|harajat|xarajat|\$|dollar|usd)/i.test(lower))) {
    if (category === 'noaniq') category = 'harajat';
    if (isDollar) currency = 'USD';

    // Parse amount
    const rawNumberMatch = cleanText.match(/(?:\$\s*)?(\d[\d\s.,]*)\s*(ming)?(?:\s*\$|\s*dollar|\s*usd)?/i);
    if (rawNumberMatch) {
      const numClean = rawNumberMatch[1].replace(/[\s.,]/g, '');
      let parsedNum = parseInt(numClean, 10);
      if (rawNumberMatch[2] || (/ming/i.test(cleanText) && currency !== 'USD')) {
        if (parsedNum < 1000) parsedNum *= 1000;
      }
      if (!isNaN(parsedNum) && parsedNum > 0) {
        amount = parsedNum;
        cleanText = cleanText
          .replace(rawNumberMatch[0], '')
          .replace(/(so'?m|som|uzs|\$|dollar|usd)/gi, '');
      }
    }

    // Classify Expense category
    const expLower = cleanText.toLowerCase();
    if (/(tushlik|nonushta|ovqat|kechki|kofe|choy|osh|kafe|restoran|bozor|supermarket|go'sht|non)/.test(expLower)) {
      expenseCategory = 'oziq_ovqat';
    } else if (/(taksi|yandex|benzin|metron?|avtobus|yo'l|zapravka)/.test(expLower)) {
      expenseCategory = 'transport';
    } else if (/(kreatin|protein|sport|zal|fitnes|trenirovka)/.test(expLower)) {
      expenseCategory = 'sport';
    } else if (/(dori|apteka|doktor|vrach|tahlil|analiz|vitamin|davolanish)/.test(expLower)) {
      expenseCategory = 'salomatlik';
    } else if (/(kitob|kurs|dars|kitoblar|repetitor|talim|universitet)/.test(expLower)) {
      expenseCategory = 'talim';
    } else {
      expenseCategory = 'shaxsiy';
    }
  }

  // 4. Quantity & Unit check for general tasks/habits (e.g. 20 sahifa, 30 daqiqa, 2 litr, 5 km)
  if (!quantity) {
    const qtyMatch = cleanText.match(/(\d+(?:\.\d+)?)\s*(sahifa|bet|varaq|daqiqa|minut|litr|km|kilometr|martta|marta|soat|ta dars|dars|gram|g)/i);
    if (qtyMatch) {
      quantity = parseFloat(qtyMatch[1]);
      unit = qtyMatch[2].toLowerCase();
      if (unit === 'bet' || unit === 'varaq') unit = 'sahifa';
      if (unit === 'minut') unit = 'daqiqa';
      if (unit === 'kilometr') unit = 'km';
      if (unit === 'martta') unit = 'marta';
    }
  }

  // 5. Date expressions in Uzbek
  const today = new Date(now);

  if (/\bindin\b/i.test(cleanText)) {
    const d = new Date(today);
    d.setDate(d.getDate() + 2);
    targetDate = formatDateYMD(d);
    cleanText = cleanText.replace(/\bindin\b/gi, '');
  } else if (/\bertaga\b/i.test(cleanText)) {
    const d = new Date(today);
    d.setDate(d.getDate() + 1);
    targetDate = formatDateYMD(d);
    cleanText = cleanText.replace(/\bertaga\b/gi, '');
  } else if (/\bbugun\b/i.test(cleanText)) {
    targetDate = formatDateYMD(today);
    cleanText = cleanText.replace(/\bbugun\b/gi, '');
  }

  // Day names: Dushanba (1) to Yakshanba (7)
  const dayNames: Record<string, number> = {
    dushanba: 1,
    seshanba: 2,
    chorshanba: 3,
    payshanba: 4,
    juma: 5,
    shanba: 6,
    yakshanba: 7,
  };

  for (const [dayName, dayNum] of Object.entries(dayNames)) {
    const regex = new RegExp(`\\b${dayName}\\b`, 'i');
    if (regex.test(cleanText)) {
      if (category === 'odat' || category === 'qabul' || /har\s+/i.test(cleanText)) {
        if (!daysOfWeek) daysOfWeek = [];
        if (!daysOfWeek.includes(dayNum)) {
          daysOfWeek.push(dayNum);
        }
      } else if (!targetDate) {
        const currentDay = today.getDay() === 0 ? 7 : today.getDay();
        let daysUntil = dayNum - currentDay;
        if (daysUntil <= 0) daysUntil += 7;
        const d = new Date(today);
        d.setDate(d.getDate() + daysUntil);
        targetDate = formatDateYMD(d);
      }
      cleanText = cleanText.replace(regex, '');
    }
  }

  // ISO or numeric dates
  const isoMatch = cleanText.match(/\b(\d{4})-(\d{2})-(\d{2})\b/);
  if (isoMatch) {
    targetDate = isoMatch[0];
    cleanText = cleanText.replace(isoMatch[0], '');
  }

  // 6. Time expressions in Uzbek
  const relMinMatch = cleanText.match(/(\d+)\s*daqiqadan\s+keyin/i);
  if (relMinMatch) {
    const mins = parseInt(relMinMatch[1], 10);
    const d = new Date(now.getTime() + mins * 60000);
    targetDate = formatDateYMD(d);
    targetTime = `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`;
    cleanText = cleanText.replace(relMinMatch[0], '');
  }

  const timeMatch = cleanText.match(/(?:soat\s*)?(\d{1,2}):(\d{2})(?:\s*da)?/i);
  if (timeMatch) {
    const hours = timeMatch[1].padStart(2, '0');
    const minutes = timeMatch[2];
    targetTime = `${hours}:${minutes}`;
    cleanText = cleanText.replace(timeMatch[0], '');
  }

  // 7. Implicit category inference
  if (category === 'noaniq') {
    if (amount) {
      category = 'harajat';
    } else if (isSupplement) {
      category = 'qabul';
    } else if (lower.includes('har kuni') || lower.includes('har dushanba') || (daysOfWeek && daysOfWeek.length > 0)) {
      category = 'odat';
    } else if (lower.includes('oy oxirigacha') || lower.includes('yil oxirigacha') || (quantity && quantity > 50)) {
      category = 'maqsad';
    } else if (lower.includes('uchrashuv') || lower.includes('suhbat') || lower.includes('majlis')) {
      category = 'uchrashuv';
    } else if (targetDate || targetTime || /(yuborish|qilish|borish|yozish|topshirish)/i.test(lower)) {
      category = 'vazifa';
    }
  }

  if (!targetDate) {
    targetDate = formatDateYMD(today);
  }

  // 8. Clean Title
  let title = cleanText
    .replace(/\s+/g, ' ')
    .replace(/^[\s,.;:-]+|[\s,.;:-]+$/g, '')
    .trim();

  if (title.length > 0) {
    title = title.charAt(0).toUpperCase() + title.slice(1);
  } else {
    if (category === 'qabul') {
      if (isCreatine) title = 'Kreatin qabul qilish';
      else if (isArginine) title = 'L-Arginin qabul qilish';
      else if (isHoneyWater) title = 'Ertalab asalli suv ichish';
      else title = 'Kunlik qo\'shimcha qabuli';
    } else if (category === 'harajat') {
      title = expenseCategory ? `${expenseCategory.toUpperCase()} xarajati` : 'Kundalik xarajat';
    } else {
      title = rawText;
    }
  }

  // 9. Human friendly explanation
  let explanationUz = '';
  if (category === 'harajat') {
    explanationUz = `💸 Xarajat: "${title}" (${amount ? formatMoney(amount, currency) : (currency === 'USD' ? '$' : "so'm")})`;
  } else if (category === 'qabul') {
    explanationUz = `💊 Kunlik qabul: "${title}" (${dosage || '1 doza'}, ${timeSlot || 'Kunlik'})`;
  } else if (category === 'vazifa') {
    explanationUz = `📌 Vazifa: "${title}" (${formatUzbekDateLabel(targetDate, targetTime)})`;
  } else if (category === 'odat') {
    explanationUz = `🔁 Odat: "${title}"`;
    if (quantity && unit) explanationUz += ` (${quantity} ${unit})`;
  } else if (category === 'maqsad') {
    explanationUz = `🎯 Maqsad: "${title}" (Muddati: ${targetDate})`;
  } else if (category === 'uchrashuv') {
    explanationUz = `🤝 Uchrashuv: "${title}" (${formatUzbekDateLabel(targetDate, targetTime)})`;
  } else {
    explanationUz = `Yozuv: "${title}"`;
  }

  return {
    category,
    title,
    date: targetDate,
    time: targetTime,
    quantity,
    unit,
    amount,
    currency,
    expenseCategory,
    dosage,
    timeSlot,
    daysOfWeek: daysOfWeek || (category === 'odat' || category === 'qabul' ? ['daily'] : null),
    confidence,
    explanationUz,
    source: 'local-rules',
  };
}

/**
 * High-level NLP parser with timeout protection and instant local fallback
 */
export async function parseUzbekIntent(text: string): Promise<ParsedItem> {
  try {
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 2000);

    const response = await fetch('/api/parse-nlp', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        text,
        currentDate: formatDateYMD(new Date()),
      }),
      signal: controller.signal,
    });
    clearTimeout(timeoutId);

    if (response.ok) {
      const data = await response.json();
      if (data.success && data.parsed) {
        return {
          category: data.parsed.category || 'vazifa',
          title: data.parsed.title || text,
          date: data.parsed.date || formatDateYMD(new Date()),
          time: data.parsed.time || null,
          quantity: data.parsed.quantity || null,
          unit: data.parsed.unit || null,
          amount: data.parsed.amount || null,
          expenseCategory: data.parsed.expenseCategory || null,
          dosage: data.parsed.dosage || null,
          timeSlot: data.parsed.timeSlot || null,
          daysOfWeek: data.parsed.daysOfWeek || null,
          confidence: data.parsed.confidence || 0.9,
          explanationUz: data.parsed.explanationUz || `${data.parsed.title}`,
          source: 'gemini-ai',
        };
      }
    }
  } catch {
    // network or dev server error; fall through to local rules
  }

  return parseUzbekTextLocal(text, new Date());
}

/**
 * Triggers Telegram Haptic Feedback with fallback to navigator.vibrate
 */
export const triggerHaptic = (
  type: 'light' | 'medium' | 'heavy' | 'success' | 'warning' | 'error' | 'selection' = 'medium'
): void => {
  if (typeof window === 'undefined') return;

  const haptic = window.Telegram?.WebApp?.HapticFeedback;

  if (haptic) {
    try {
      if (type === 'selection') {
        haptic.selectionChanged();
        return;
      }
      if (type === 'success' || type === 'warning' || type === 'error') {
        haptic.notificationOccurred(type);
        return;
      }
      haptic.impactOccurred(type);
      return;
    } catch {
      // Fallback
    }
  }

  // Web vibration fallback for supported mobile browsers
  if ('vibrate' in navigator) {
    try {
      if (type === 'success') navigator.vibrate([20, 50, 30]);
      else if (type === 'error') navigator.vibrate([40, 40, 40]);
      else if (type === 'heavy') navigator.vibrate(35);
      else navigator.vibrate(15);
    } catch {
      // Ignore vibration error if not supported/permitted
    }
  }
};

    setEditFrequency(h.frequency || 'daily');
    setEditTargetDaysPerWeek(h.targetDaysPerWeek || 3);
    setEditSelectedDays(h.daysOfWeek || [1]);
    setEditHasQuantity(!!h.hasQuantity);
    setEditTargetQuantity(h.targetQuantity || 1);
    setEditUnit(h.unit || 'marta');
    setEditLinkedGoalId(h.linkedGoalId || '');
  };

  const handleUpdateHabit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!editingHabit || !editTitle.trim()) return;

    updateHabit(editingHabit.id, {
      title: editTitle.trim(),
      habitType: editHabitType,
      dosage: editHabitType === 'supplement_health' ? editDosage.trim() : undefined,
      timeSlot: editHabitType === 'supplement_health' ? editTimeSlot.trim() : undefined,
      frequency: editFrequency,
      targetDaysPerWeek: editFrequency === 'weekly_target' ? editTargetDaysPerWeek : undefined,
      daysOfWeek: editFrequency === 'specific_days' ? editSelectedDays : undefined,
      hasQuantity: editHasQuantity,
      targetQuantity: editHasQuantity ? Number(editTargetQuantity) || 1 : undefined,
      unit: editHasQuantity ? editUnit : undefined,
      linkedGoalId: editLinkedGoalId || undefined,
    });

    setEditingHabit(null);
  };

  const toggleEditDay = (dayNum: number) => {
    setEditSelectedDays(prev =>
      prev.includes(dayNum) ? (prev.length > 1 ? prev.filter(d => d !== dayNum) : prev) : [...prev, dayNum].sort()
    );
  };

  const daysLabels = [
    { num: 1, label: 'Du' },
    { num: 2, label: 'Se' },
    { num: 3, label: 'Chor' },
    { num: 4, label: 'Pay' },
    { num: 5, label: 'Jum' },
    { num: 6, label: 'Shan' },
    { num: 7, label: 'Yak' },
  ];

  const toggleDaySelection = (dayNum: number) => {
    setSelectedDays(prev => {
      if (prev.includes(dayNum)) {
        if (prev.length === 1) return prev; // Keep at least one day selected
        return prev.filter(d => d !== dayNum);
      } else {
        return [...prev, dayNum].sort();
      }
    });
  };

  const handleCreateHabit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!title.trim()) return;

    let finalDays: number[] = [1, 2, 3, 4, 5, 6, 7];
    if (frequency === 'specific_days') {
      finalDays = selectedDays.length > 0 ? selectedDays : [1];
    } else if (frequency === 'weekly_target') {
      finalDays = [];
    }

    addHabit({
      title: title.trim(),
      category: 'odat',
      habitType,
      dosage: habitType === 'supplement_health' && dosage.trim() ? dosage.trim() : undefined,
      timeSlot: habitType === 'supplement_health' && timeSlot.trim() ? timeSlot.trim() : undefined,
      frequency,
      targetDaysPerWeek: frequency === 'weekly_target' ? targetDaysPerWeek : undefined,
      daysOfWeek: finalDays,
      hasQuantity,
      targetQuantity: hasQuantity ? Number(targetQuantity) : undefined,
      unit: hasQuantity && unit.trim() ? unit.trim() : undefined,
      linkedGoalId: linkedGoalId || undefined,
    });

    // Reset Form
    setTitle('');
    setDosage('');
    setTimeSlot('');
    setFrequency('daily');
    setTargetDaysPerWeek(3);
    setSelectedDays([1]);
    setHasQuantity(false);
    setTargetQuantity(1);
    setUnit('marta');
    setLinkedGoalId('');
    setIsAddModalOpen(false);
  };

  // Generate last 7 days keys for weekly matrix
  const getLast7Days = () => {
    const list: string[] = [];
    for (let i = 6; i >= 0; i--) {
      const d = new Date();
      d.setDate(d.getDate() - i);
      list.push(formatDateYMD(d));
    }
    return list;
  };
  const last7Days = getLast7Days();

  const filteredHabits = habits.filter(h => {
    if (h.isArchived) return false;
    if (filterType === 'health') return h.habitType === 'supplement_health';
    if (filterType === 'general') return h.habitType !== 'supplement_health';
    return true;
  });

  const getFrequencyLabel = (habit: typeof habits[0]) => {
    if (habit.frequency === 'daily') return 'Har kuni';
    if (habit.frequency === 'weekly_target') return `Haftada ${habit.targetDaysPerWeek || 3} marta`;
    if (habit.frequency === 'specific_days') {
      if (habit.daysOfWeek?.length === 1 && habit.daysOfWeek[0] === 1) {
        return 'Har dushanba kuni';
      }
      const dayNames = (habit.daysOfWeek || [])
        .map(d => daysLabels.find(l => l.num === d)?.label)
        .filter(Boolean)
        .join(', ');
      return dayNames ? `Haftaning: ${dayNames}` : 'Muayyan kunlarda';
    }
    return 'Muntazam';
  };

  return (
    <div className="space-y-4 pb-6">
      {/* Top Header */}
      <div className="flex items-center justify-between">
        <div>
          <h2 className="text-lg font-bold text-slate-100">Odatlar</h2>
          <p className="text-xs text-slate-400">Kunlik va haftalik muntazam intizom</p>
        </div>
        <button
          onClick={() => setIsAddModalOpen(true)}
          className="bg-sky-600 hover:bg-sky-500 text-white font-medium text-xs px-3.5 py-2 rounded-xl flex items-center gap-1.5 transition-all shadow-sm active:scale-95 min-h-[40px]"
        >
          <Plus className="w-4 h-4" />
          <span>Yangi odat</span>
        </button>
      </div>

      {/* Filter Tabs */}
      <div className="flex items-center gap-1.5 overflow-x-auto no-scrollbar py-0.5">
        <button
          onClick={() => setFilterType('all')}
          className={`text-xs font-medium px-3.5 py-1.5 rounded-full whitespace-nowrap transition-all ${
            filterType === 'all'
              ? 'bg-sky-600 text-white shadow-sm'
              : 'bg-slate-800 text-slate-400 hover:text-slate-200 border border-slate-700/60'
          }`}
        >
          Barchasi ({habits.filter(h => !h.isArchived).length})
        </button>
        <button
          onClick={() => setFilterType('health')}
          className={`text-xs font-medium px-3.5 py-1.5 rounded-full whitespace-nowrap transition-all ${
            filterType === 'health'
              ? 'bg-emerald-600 text-white shadow-sm'
              : 'bg-slate-800 text-slate-400 hover:text-slate-200 border border-slate-700/60'
          }`}
        >
          Salomatlik
        </button>
        <button
          onClick={() => setFilterType('general')}
          className={`text-xs font-medium px-3.5 py-1.5 rounded-full whitespace-nowrap transition-all ${
            filterType === 'general'
              ? 'bg-sky-600 text-white shadow-sm'
              : 'bg-slate-800 text-slate-400 hover:text-slate-200 border border-slate-700/60'
          }`}
        >
          Umumiy odatlar
        </button>
      </div>

      {/* Habits List */}
      <div className="space-y-3">
        {filteredHabits.length === 0 ? (
          <div className="text-center py-10 px-4 bg-slate-900/40 rounded-2xl border border-dashed border-slate-800 text-slate-400 text-xs space-y-2">
            <p className="text-slate-300 font-semibold">Hozircha odatlar mavjud emas</p>
            <p className="text-slate-400 text-[11px]">
              Har kungi yoki haftaning ma'lum kunlaridagi muntazam odatlaringizni kiriting!
            </p>
            <div>
              <button
                onClick={() => setIsAddModalOpen(true)}
                className="mt-1 inline-flex items-center gap-1.5 px-3.5 py-2 rounded-xl bg-sky-600 hover:bg-sky-500 text-white font-semibold text-xs shadow-sm transition-all active:scale-95"
              >
                <Plus className="w-3.5 h-3.5" />
                <span>Birinchi odatni qo'shish</span>
              </button>
            </div>
          </div>
        ) : (
          filteredHabits.map(habit => {
            const todayRecord = habit.history[todayStr] || { completed: false, quantity: 0 };
            const isDone = todayRecord.completed;
            const linkedGoal = goals.find(g => g.id === habit.linkedGoalId);
            const isHealth = habit.habitType === 'supplement_health';

            return (
              <div
                key={habit.id}
                className={`bg-slate-900/80 border rounded-2xl p-4 shadow-sm hover:border-slate-600 transition-all space-y-3 ${
                  isHealth ? 'border-emerald-500/30' : 'border-slate-700/80'
                }`}
              >
                {/* Habit Header & Checkbox */}
                <div className="flex items-start justify-between gap-3">
                  <div className="flex items-start gap-3 flex-1 min-w-0">
                    <button
                      onClick={() => toggleHabitToday(habit.id)}
                      className="mt-0.5 shrink-0"
                    >
                      {isDone ? (
                        <CheckCircle2 className="w-6 h-6 text-emerald-400" />
                      ) : (
                        <Circle className="w-6 h-6 text-slate-500 hover:text-sky-400 transition-colors" />
                      )}
                    </button>

                    <div className="min-w-0 flex-1">
                      <div className="flex items-center gap-2">
                        <h4
                          className={`text-sm font-semibold truncate ${
                            isDone ? 'text-slate-400' : 'text-slate-100'
                          }`}
                        >
                          {habit.title}
                        </h4>
                        <span className="flex items-center gap-1 text-xs font-semibold px-2 py-0.5 rounded-full bg-amber-500/10 text-amber-400 border border-amber-500/20 shrink-0">
                          <Flame className="w-3.5 h-3.5 fill-amber-400" />
                          {habit.streak} kun
                        </span>
                      </div>

                      {/* Frequency Badge */}
                      <div className="flex flex-wrap items-center gap-2 mt-1.5 text-xs">
                        <span className="px-2 py-0.5 rounded-md bg-slate-800 text-slate-300 border border-slate-700/80 flex items-center gap-1 text-[11px] font-medium">
                          <Repeat className="w-3 h-3 text-sky-400" />
                          {getFrequencyLabel(habit)}
                        </span>

                        {isHealth && habit.dosage && (
                          <span className="px-2 py-0.5 rounded-md bg-emerald-500/15 text-emerald-300 font-medium border border-emerald-500/25 flex items-center gap-1 text-[11px]">
                            <Pill className="w-3 h-3" />
                            {habit.dosage}
                          </span>
                        )}

                        {isHealth && habit.timeSlot && (
                          <span className="px-2 py-0.5 rounded-md bg-slate-800 text-slate-300 border border-slate-700 flex items-center gap-1 text-[11px]">
                            <Clock className="w-3 h-3 text-sky-400" />
                            {habit.timeSlot}
                          </span>
                        )}
                      </div>

                      {/* Linked Goal Badge */}
                      {linkedGoal && (
                        <div className="inline-flex items-center gap-1 text-[11px] text-sky-400 bg-sky-500/10 border border-sky-500/20 px-2 py-0.5 rounded-md mt-1.5 font-medium">
                          <Target className="w-3 h-3" />
                          <span>Maqsad: {linkedGoal.title}</span>
                        </div>
                      )}
                    </div>
                  </div>

                  <div className="flex items-center gap-1">
                    <button
                      onClick={() => openEditModal(habit)}
                      className="text-slate-400 hover:text-sky-400 p-1.5 rounded-lg hover:bg-slate-800 transition-colors"
                      title="O'zgartirish / Tahrirlash"
                    >
                      <Pencil className="w-4 h-4" />
                    </button>
                    <button
                      onClick={() => deleteHabit(habit.id)}
                      className="text-slate-500 hover:text-rose-400 p-1.5 rounded-lg hover:bg-slate-800 transition-colors"
                      title="O'chirish"
                    >
                      <Trash2 className="w-4 h-4" />
                    </button>
                  </div>
                </div>

                {/* Quantity Logger */}
                {habit.hasQuantity && habit.targetQuantity && (
                  <div className="bg-slate-800/80 rounded-xl p-3 border border-slate-700/60 flex items-center justify-between gap-3">
                    <div>
                      <div className="flex items-center gap-2 text-xs">
                        <span className="font-semibold text-slate-200">Bugungi miqdor:</span>
                        <span className="text-emerald-400 font-bold">
                          {todayRecord.quantity || 0} / {habit.targetQuantity} {habit.unit}
                        </span>
                      </div>
                      <div className="w-32 sm:w-44 bg-slate-900 rounded-full h-1.5 overflow-hidden mt-1.5">
                        <div
                          className="bg-emerald-400 h-full rounded-full transition-all"
                          style={{
                            width: `${Math.min(
                              100,
                              Math.round(((todayRecord.quantity || 0) / habit.targetQuantity) * 100)
                            )}%`,
                          }}
                        />
                      </div>
                    </div>

                    <div className="flex items-center gap-1 shrink-0">
                      <button
                        onClick={() => logHabitQuantity(habit.id, -1)}
                        className="px-2 py-1 text-xs text-slate-300 bg-slate-700 hover:bg-slate-600 rounded-lg transition-colors"
                        title="-1"
                      >
                        -1
                      </button>
                      <button
                        onClick={() => logHabitQuantity(habit.id, 1)}
                        className="px-2.5 py-1 text-xs font-semibold text-white bg-emerald-600 hover:bg-emerald-500 rounded-lg transition-colors shadow-sm"
                        title="+1"
                      >
                        +1 {habit.unit}
                      </button>
                    </div>
                  </div>
                )}

                {/* Last 7 Days Mini Dot Grid */}
                <div className="pt-2 border-t border-slate-800 flex items-center justify-between text-xs">
                  <span className="text-slate-400 text-[11px]">Oxirgi 7 kun:</span>
                  <div className="flex items-center gap-1.5">
                    {last7Days.map(dateKey => {
                      const dayRec = habit.history[dateKey];
                      const dayDone = dayRec?.completed;
                      const [,, d] = dateKey.split('-');

                      return (
                        <div
                          key={dateKey}
                          title={`${dateKey}: ${dayDone ? 'Bajarilgan' : 'Bajarilmagan'}`}
                          className={`w-6 h-6 rounded-lg flex items-center justify-center text-[10px] font-medium transition-all ${
                            dayDone
                              ? 'bg-emerald-500 text-white font-bold'
                              : 'bg-slate-800 text-slate-500 border border-slate-700'
                          }`}
                        >
                          {parseInt(d, 10)}
                        </div>
                      );
                    })}
                  </div>
                </div>
              </div>
            );
          })
        )}
      </div>

      {/* Add Habit Modal */}
      {isAddModalOpen && (
        <div className="fixed inset-0 z-50 bg-black/70 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-slate-900 border border-slate-800 rounded-3xl w-full max-w-md p-5 shadow-2xl space-y-4 animate-in fade-in zoom-in-95 duration-200 max-h-[90vh] overflow-y-auto">
            <div className="flex items-center justify-between pb-2 border-b border-slate-800">
              <h3 className="font-bold text-base text-slate-100">Yangi Odat Qo'shish</h3>
              <button
                onClick={() => setIsAddModalOpen(false)}
                className="text-slate-400 hover:text-white p-1 rounded-lg hover:bg-slate-800"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleCreateHabit} className="space-y-3.5">
              {/* Type selector */}
              <div>
                <label className="block text-xs font-medium text-slate-300 mb-1.5">Odat toifasi</label>
                <div className="grid grid-cols-2 gap-2">
                  <button
                    type="button"
                    onClick={() => setHabitType('general')}
                    className={`py-2 rounded-xl text-xs font-medium border flex items-center justify-center gap-1.5 transition-colors ${
                      habitType === 'general'
                        ? 'bg-sky-600 text-white border-transparent'
                        : 'bg-slate-800 text-slate-300 border-slate-700'
                    }`}
                  >
                    Umumiy odat
                  </button>
                  <button
                    type="button"
                    onClick={() => setHabitType('supplement_health')}
                    className={`py-2 rounded-xl text-xs font-medium border flex items-center justify-center gap-1.5 transition-colors ${
                      habitType === 'supplement_health'
                        ? 'bg-emerald-600 text-white border-transparent'
                        : 'bg-slate-800 text-slate-300 border-slate-700'
                    }`}
                  >
                    <Pill className="w-3.5 h-3.5" />
                    Salomatlik
                  </button>
                </div>
              </div>

              {/* Title */}
              <div>
                <label className="block text-xs font-medium text-slate-300 mb-1">
                  Nomi *
                </label>
                <input
                  type="text"
                  value={title}
                  onChange={e => setTitle(e.target.value)}
                  placeholder="Odat nomini kiriting..."
                  required
                  className="w-full bg-slate-800 border border-slate-700 rounded-xl px-3.5 py-2.5 text-sm text-slate-100 placeholder-slate-500 focus:outline-none focus:border-sky-500"
                />
              </div>

              {/* Frequency Selector: Har kuni, Haftada N marta, Hafta kunlari */}
              <div>
                <label className="block text-xs font-medium text-slate-300 mb-1.5">
                  Takrorlanish chastotasi *
                </label>
                <div className="grid grid-cols-3 gap-1.5 mb-2.5">
                  <button
                    type="button"
                    onClick={() => setFrequency('daily')}
                    className={`py-2 px-1 text-xs rounded-xl font-medium border text-center transition-all ${
                      frequency === 'daily'
                        ? 'bg-sky-600 text-white border-transparent shadow-sm'
                        : 'bg-slate-800 text-slate-300 border-slate-700 hover:text-white'
                    }`}
                  >
                    Har kuni
                  </button>
                  <button
                    type="button"
                    onClick={() => setFrequency('weekly_target')}
                    className={`py-2 px-1 text-xs rounded-xl font-medium border text-center transition-all ${
                      frequency === 'weekly_target'
                        ? 'bg-sky-600 text-white border-transparent shadow-sm'
                        : 'bg-slate-800 text-slate-300 border-slate-700 hover:text-white'
                    }`}
                  >
                    Haftada N marta
                  </button>
                  <button
                    type="button"
                    onClick={() => setFrequency('specific_days')}
                    className={`py-2 px-1 text-xs rounded-xl font-medium border text-center transition-all ${
                      frequency === 'specific_days'
                        ? 'bg-sky-600 text-white border-transparent shadow-sm'
                        : 'bg-slate-800 text-slate-300 border-slate-700 hover:text-white'
                    }`}
                  >
                    Hafta kunlari
                  </button>
                </div>

                {/* Sub-options for weekly_target */}
                {frequency === 'weekly_target' && (
                  <div className="bg-slate-800/60 p-3 rounded-2xl border border-slate-700/70 flex items-center justify-between">
                    <span className="text-xs text-slate-300 font-medium">Haftasiga necha marta?</span>
                    <div className="flex items-center gap-1.5">
                      {[1, 2, 3, 4, 5, 6].map(num => (
                        <button
                          key={num}
                          type="button"
                          onClick={() => setTargetDaysPerWeek(num)}
                          className={`w-8 h-8 rounded-lg text-xs font-bold transition-all ${
                            targetDaysPerWeek === num
                              ? 'bg-sky-600 text-white shadow-sm'
                              : 'bg-slate-700 text-slate-300 hover:bg-slate-600'
                          }`}
                        >
                          {num}
                        </button>
                      ))}
                    </div>
                  </div>
                )}

                {/* Sub-options for specific_days (e.g. Faqat har dushanba) */}
                {frequency === 'specific_days' && (
                  <div className="bg-slate-800/60 p-3 rounded-2xl border border-slate-700/70 space-y-2">
                    <div className="flex items-center justify-between">
                      <span className="text-xs text-slate-300 font-medium">Qaysi kunlarda bajariladi?</span>
                      {selectedDays.length === 1 && selectedDays[0] === 1 && (
                        <span className="text-[11px] text-sky-400 font-semibold bg-sky-500/10 px-2 py-0.5 rounded-md border border-sky-500/20">
                          Faqat har dushanba
                        </span>
                      )}
                    </div>
                    <div className="grid grid-cols-7 gap-1">
                      {daysLabels.map(d => {
                        const isSel = selectedDays.includes(d.num);
                        return (
                          <button
                            key={d.num}
                            type="button"
                            onClick={() => toggleDaySelection(d.num)}
                            className={`py-2 text-[11px] font-semibold rounded-lg border transition-all ${
                              isSel
                                ? 'bg-sky-600 border-sky-500 text-white shadow-sm'
                                : 'bg-slate-800 border-slate-700 text-slate-400 hover:text-slate-200'
                            }`}
                          >
                            {d.label}
                          </button>
                        );
                      })}
                    </div>
                  </div>
                )}
              </div>

              {/* Supplement specific fields (Only if Salomatlik is chosen) */}
              {habitType === 'supplement_health' && (
                <div className="grid grid-cols-2 gap-3 bg-slate-800/50 p-3 rounded-2xl border border-slate-700/60">
                  <div>
                    <label className="block text-xs font-medium text-slate-300 mb-1">
                      Miqdor / Doza
                    </label>
                    <input
                      type="text"
                      value={dosage}
                      onChange={e => setDosage(e.target.value)}
                      placeholder="Miqdor (masalan: 1 tabletka)"
                      className="w-full bg-slate-800 border border-slate-700 rounded-xl px-3 py-2 text-xs text-slate-100 focus:outline-none focus:border-emerald-500"
                    />
                  </div>
                  <div>
                    <label className="block text-xs font-medium text-slate-300 mb-1">
                      Qabul vaqti
                    </label>
                    <select
                      value={timeSlot}
                      onChange={e => setTimeSlot(e.target.value)}
                      className="w-full bg-slate-800 border border-slate-700 rounded-xl px-2 py-2 text-xs text-slate-100 focus:outline-none focus:border-emerald-500"
                    >
                      <option value="">Tanlang...</option>
                      <option value="Ertalab nahorga">Ertalab nahorga</option>
                      <option value="Kun davomida">Kun davomida</option>
                      <option value="Ovqatdan oldin">Ovqatdan oldin</option>
                      <option value="Ovqatdan keyin">Ovqatdan keyin</option>
                      <option value="Kechqurun yotishdan oldin">Kechqurun yotishdan oldin</option>
                    </select>
                  </div>
                </div>
              )}

              {/* Quantity Tracking Toggle */}
              <div className="pt-2 border-t border-slate-800">
                <div className="flex items-center justify-between">
                  <div>
                    <label className="text-xs font-medium text-slate-200">
                      Raqam bilan hisoblash
                    </label>
                    <p className="text-[11px] text-slate-400">
                      Miqdor va o'lchov birligini belgilash
                    </p>
                  </div>
                  <input
                    type="checkbox"
                    checked={hasQuantity}
                    onChange={e => setHasQuantity(e.target.checked)}
                    className="w-4 h-4 rounded text-sky-600 accent-sky-500 cursor-pointer"
                  />
                </div>

                {hasQuantity && (
                  <div className="grid grid-cols-2 gap-3 mt-3">
                    <div>
                      <label className="block text-xs font-medium text-slate-300 mb-1">
                        Kunlik maqsad (son)
                      </label>
                      <input
                        type="number"
                        min="1"
                        value={targetQuantity}
                        onChange={e => setTargetQuantity(Number(e.target.value))}
                        className="w-full bg-slate-800 border border-slate-700 rounded-xl px-3 py-2 text-xs text-slate-100 focus:outline-none focus:border-sky-500"
                      />
                    </div>
                    <div>
                      <label className="block text-xs font-medium text-slate-300 mb-1">
                        Birlik
                      </label>
                      <input
                        type="text"
                        value={unit}
                        onChange={e => setUnit(e.target.value)}
                        placeholder="marta, sahifa, daqiqa"
                        className="w-full bg-slate-800 border border-slate-700 rounded-xl px-3 py-2 text-xs text-slate-100 focus:outline-none focus:border-sky-500"
                      />
                    </div>
                  </div>
                )}
              </div>

              {/* Link to Goal Selector */}
              {goals.length > 0 && (
                <div>
                  <label className="block text-xs font-medium text-slate-300 mb-1">
                    Katta Maqsadga bog'lash (Ixtiyoriy)
                  </label>
                  <select

            let percent = 0;
            let remaining = 0;
            let isAchieved = false;

            if (isDecrease) {
              const start = goal.startValue || 90;
              const target = goal.targetValue;
              const current = goal.currentValue;
              const totalToLose = Math.max(0.1, start - target);
              const lostSoFar = Math.max(0, start - current);
              percent = Math.min(100, Math.round((lostSoFar / totalToLose) * 100));
              remaining = Math.max(0, current - target);
              isAchieved = current <= target;
            } else {
              percent = Math.min(100, Math.round((goal.currentValue / goal.targetValue) * 100));
              remaining = Math.max(0, goal.targetValue - goal.currentValue);
              isAchieved = goal.currentValue >= goal.targetValue;
            }

            const linkedHabitsList = habits.filter(h => h.linkedGoalId === goal.id && !h.isArchived);
            const internalHabits = goal.internalHabits || [];
            const progressLogs = goal.progressLogs || [];
            const isLogsExpanded = expandedLogsGoalId === goal.id;

            return (
              <div
                key={goal.id}
                className={`bg-slate-900/80 border rounded-2xl p-4 shadow-sm space-y-3.5 transition-all ${
                  isAchieved
                    ? 'border-emerald-500/50 bg-emerald-950/20'
                    : 'border-slate-700/80 hover:border-slate-600'
                }`}
              >
                {/* Header & Status */}
                <div className="flex items-start justify-between gap-3">
                  <div className="flex-1 min-w-0">
                    <div className="flex items-center gap-2">
                      {isDecrease ? (
                        <Scale className="w-4 h-4 text-rose-400 shrink-0" />
                      ) : (
                        <TrendingUp className="w-4 h-4 text-sky-400 shrink-0" />
                      )}
                      <h4 className="text-sm font-bold text-slate-100 truncate">{goal.title}</h4>
                      {isAchieved && (
                        <span className="flex items-center gap-1 px-2 py-0.5 rounded-full bg-emerald-500/20 text-emerald-400 text-[10px] font-semibold border border-emerald-500/30">
                          <Award className="w-3 h-3" />
                          Erishildi!
                        </span>
                      )}
                    </div>

                    <div className="flex items-center gap-2 text-xs text-slate-400 mt-1">
                      <span className="flex items-center gap-1">
                        <Calendar className="w-3.5 h-3.5 text-sky-400" />
                        Muddat: {goal.deadline}
                      </span>
                      <span>•</span>
                      <span className={daysLeft > 0 ? 'text-amber-400' : 'text-rose-400'}>
                        {daysLeft > 0 ? `${daysLeft} kun qoldi` : 'Muddat yetib keldi'}
                      </span>
                    </div>

                    {goal.notes && (
                      <p className="text-[11px] text-slate-400 mt-1 bg-slate-800/40 px-2 py-1 rounded-md border border-slate-800 inline-block">
                        💡 {goal.notes}
                      </p>
                    )}
                  </div>

                  <div className="flex items-center gap-1">
                    {goal.isArchived ? (
                      <>
                        <button
                          onClick={() => restoreGoal(goal.id)}
                          className="text-xs bg-sky-500/20 hover:bg-sky-500/30 text-sky-300 px-2.5 py-1 rounded-lg border border-sky-500/30 flex items-center gap-1 transition-colors"
                          title="Qayta tiklash"
                        >
                          <RotateCcw className="w-3.5 h-3.5" />
                          <span>Tiklash</span>
                        </button>
                        <button
                          onClick={() => {
                            if (window.confirm("Ushbu maqsadni butunlay o'chirasizmi?")) {
                              deleteGoal(goal.id, true);
                            }
                          }}
                          className="text-slate-500 hover:text-rose-400 p-1.5 rounded-lg transition-colors"
                          title="Butunlay o'chirish"
                        >
                          <Trash2 className="w-4 h-4" />
                        </button>
                      </>
                    ) : (
                      <div className="flex items-center gap-1">
                        <button
                          onClick={() => openEditModal(goal)}
                          className="text-slate-400 hover:text-sky-400 p-1.5 rounded-lg hover:bg-slate-800 transition-colors"
                          title="O'zgartirish / Tahrirlash"
                        >
                          <Pencil className="w-4 h-4" />
                        </button>
                        <button
                          onClick={() => deleteGoal(goal.id, false)}
                          className="text-slate-500 hover:text-rose-400 p-1.5 rounded-lg hover:bg-slate-800 transition-colors"
                          title="Arxivga o'tkazish (saqlanib qoladi)"
                        >
                          <Trash2 className="w-4 h-4" />
                        </button>
                      </div>
                    )}
                  </div>
                </div>

                {/* COUNTDOWN & REMAINING CARD (User Request: 300 - 30 = 270 qoldi) */}
                <div className="bg-gradient-to-r from-sky-950/70 via-slate-900 to-indigo-950/60 border border-sky-500/30 rounded-2xl p-3.5 flex flex-wrap items-center justify-between gap-3">
                  <div>
                    <span className="text-[11px] font-semibold text-sky-300 uppercase tracking-wider block">
                      {isDecrease ? 'Ozish / Kamaytirish qoldig\'i:' : 'Maqsadga erishish qoldig\'i:'}
                    </span>
                    <div className="text-base sm:text-lg font-bold text-white flex items-center gap-2 mt-0.5">
                      {isAchieved ? (
                        <span className="text-emerald-400 font-extrabold flex items-center gap-1.5">
                          <Award className="w-5 h-5 text-amber-400" />
                          Maqsadga to'liq erishildi! 🎉
                        </span>
                      ) : isDecrease ? (
                        <>
                          <span className="text-rose-400 font-black">{remaining.toFixed(1)} {goal.unit}</span>
                          <span className="text-slate-300 text-xs font-normal">
                            kamaytirish qoldi ({goal.currentValue} {goal.unit} → {goal.targetValue} {goal.unit})
                          </span>
                        </>
                      ) : (
                        <>
                          <span className="text-sky-400 font-black">{remaining} {goal.unit}</span>
                          <span className="text-slate-300 text-xs font-normal">
                            bajarilishi qoldi ({goal.targetValue} - {goal.currentValue} = {remaining})
                          </span>
                        </>
                      )}
                    </div>
                  </div>

                  <div className="text-right">
                    <span className="text-[11px] text-slate-400 block">Joriy holat:</span>
                    <span className="text-xs sm:text-sm font-bold text-emerald-400">
                      {goal.currentValue} / {goal.targetValue} {goal.unit} ({percent}%)
                    </span>
                  </div>
                </div>

                {/* Progress Bar */}
                <div className="w-full bg-slate-800 rounded-full h-2.5 overflow-hidden">
                  <div
                    className={`h-full rounded-full transition-all duration-500 ${
                      isAchieved
                        ? 'bg-gradient-to-r from-emerald-500 to-teal-400'
                        : isDecrease
                        ? 'bg-gradient-to-r from-amber-500 via-rose-500 to-emerald-500'
                        : 'bg-gradient-to-r from-sky-500 via-indigo-500 to-emerald-500'
                    }`}
                    style={{ width: `${percent}%` }}
                  />
                </div>

                {/* GOAL-INTERNAL HABITS / SUB-HABITS (User Request: 90kg -> 85kg ichida alohida odatlar) */}
                <div className="bg-slate-800/60 rounded-2xl p-3 border border-slate-700/60 space-y-2.5">
                  <div className="flex items-center justify-between">
                    <span className="text-xs font-bold text-slate-200 flex items-center gap-1.5">
                      <ListTodo className="w-3.5 h-3.5 text-amber-400" />
                      Maqsad uchun alohida odatlar & rutinalar ({internalHabits.length})
                    </span>
                    {!goal.isArchived && (
                      <button
                        onClick={() =>
                          setActiveGoalForHabitAdd(
                            activeGoalForHabitAdd === goal.id ? null : goal.id
                          )
                        }
                        className="text-[11px] text-sky-400 hover:text-sky-300 flex items-center gap-1 font-semibold"
                      >
                        <Plus className="w-3 h-3" />
                        <span>Odat qo'shish</span>
                      </button>
                    )}
                  </div>

                  {/* Inline Form to add internal habit */}
                  {activeGoalForHabitAdd === goal.id && (
                    <form
                      onSubmit={e => handleAddInternalHabitSubmit(goal.id, e)}
                      className="bg-slate-900/90 p-2.5 rounded-xl border border-sky-500/40 space-y-2 animate-in fade-in"
                    >
                      <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
                        <input
                          type="text"
                          value={internalHabitTitle}
                          onChange={e => setInternalHabitTitle(e.target.value)}
                          placeholder="Masalan: Haftada 3 kun zal, 5g kreatin..."
                          required
                          className="bg-slate-800 border border-slate-700 rounded-lg px-2.5 py-1.5 text-xs text-slate-100 placeholder-slate-500 focus:outline-none focus:border-sky-500"
                        />
                        <select
                          value={internalHabitFreq}
                          onChange={e => setInternalHabitFreq(e.target.value)}
                          className="bg-slate-800 border border-slate-700 rounded-lg px-2.5 py-1.5 text-xs text-slate-200 focus:outline-none focus:border-sky-500"
                        >
                          <option value="Har kuni">Har kuni</option>
                          <option value="Haftada 3 kun">Haftada 3 kun</option>
                          <option value="Har kuni ertalab (Nahorga)">Har kuni ertalab (Nahorga)</option>
                          <option value="Kechqurun">Kechqurun</option>
                        </select>
                      </div>
                      <div className="flex justify-end gap-1.5">
                        <button
                          type="button"
                          onClick={() => setActiveGoalForHabitAdd(null)}
                          className="px-2.5 py-1 rounded-lg text-[11px] text-slate-400 hover:text-slate-200"
                        >
                          Bekor qilish
                        </button>
                        <button
                          type="submit"
                          className="px-3 py-1 rounded-lg text-[11px] bg-sky-600 hover:bg-sky-500 text-white font-semibold"
                        >
                          Qo'shish
                        </button>
                      </div>
                    </form>
                  )}

                  {internalHabits.length === 0 ? (
                    <p className="text-[11px] text-slate-400 italic">
                      Hozircha ichki odatlar kiritilmagan. Maqsadga erishish uchun kichik kundalik odatlarni belgilang (masalan: zalga chiqish, kreatin, nahorga suv).
                    </p>
                  ) : (
                    <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
                      {internalHabits.map(ih => (
                        <div
                          key={ih.id}
                          className={`flex items-center justify-between p-2 rounded-xl border text-xs transition-colors ${
                            ih.completedToday
                              ? 'bg-emerald-950/30 border-emerald-500/30 text-emerald-200'
                              : 'bg-slate-900/60 border-slate-800 text-slate-300'
                          }`}
                        >
                          <button
                            onClick={() => toggleGoalInternalHabit(goal.id, ih.id)}
                            className="flex items-center gap-2 text-left flex-1 min-w-0"
                          >
                            {ih.completedToday ? (
                              <CheckCircle2 className="w-4 h-4 text-emerald-400 shrink-0" />
                            ) : (
                              <Circle className="w-4 h-4 text-slate-500 shrink-0" />
                            )}
                            <div className="min-w-0 flex-1">
                              <span className={`truncate block font-medium ${ih.completedToday ? 'line-through text-slate-400' : ''}`}>
                                {ih.title}
                              </span>
                              <span className="text-[10px] text-slate-400">
                                {ih.targetFrequency} • 🔥 {ih.streak || 0} kun
                              </span>
                            </div>
                          </button>
                          {!goal.isArchived && (
                            <button
                              onClick={() => deleteGoalInternalHabit(goal.id, ih.id)}
                              className="text-slate-600 hover:text-rose-400 p-1 rounded transition-colors"
                              title="O'chirish"
                            >
                              <X className="w-3 h-3" />
                            </button>
                          )}
                        </div>
                      ))}
                    </div>
                  )}
                </div>

                {/* Action Buttons: Log Progress and View Logs */}
                {!goal.isArchived && (
                  <div className="pt-1 flex flex-wrap items-center justify-between gap-2">
                    <button
                      onClick={() =>
                        setExpandedLogsGoalId(isLogsExpanded ? null : goal.id)
                      }
                      className="text-xs text-slate-400 hover:text-slate-200 flex items-center gap-1 font-medium"
                    >
                      <span>Oxirgi qaydlar va tarix ({progressLogs.length})</span>
                      {isLogsExpanded ? <ChevronUp className="w-3.5 h-3.5" /> : <ChevronDown className="w-3.5 h-3.5" />}
                    </button>

                    <button
                      onClick={() => {
                        setSelectedGoalIdForLog(goal.id);
                        setLogAmount(isDecrease ? 1 : 30);
                        setLogNote('');
                      }}
                      className="bg-sky-600 hover:bg-sky-500 text-white font-semibold text-xs px-3.5 py-2 rounded-xl flex items-center gap-1.5 transition-all active:scale-95 shadow-sm"
                    >
                      <Plus className="w-3.5 h-3.5" />
                      <span>{isDecrease ? "+ Vazn o'lchovi kiritish" : "+ Aniq natija kiritish"}</span>
                    </button>
                  </div>
                )}

                {/* Progress Logs History Accordion (User Request: Xato yozilsa bekor qilish) */}
                {isLogsExpanded && (
                  <div className="bg-slate-950/60 rounded-xl p-3 border border-slate-800 space-y-2 animate-in fade-in">
                    <div className="flex items-center justify-between text-[11px] text-slate-400 border-b border-slate-800/80 pb-1.5">
                      <span className="font-semibold text-slate-300">Bajarilgan ishlar jurnali:</span>
                      <span>Xato kiritilsa bekor qilishingiz mumkin</span>
                    </div>

                    {progressLogs.length === 0 ? (
                      <p className="text-[11px] text-slate-500 italic py-2 text-center">
                        Hozircha natija qaydlari mavjud emas.
                      </p>
                    ) : (
                      <div className="space-y-1.5 max-h-48 overflow-y-auto pr-1">
                        {progressLogs.map(log => (
                          <div
                            key={log.id}
                            className="flex items-center justify-between bg-slate-900/80 p-2 rounded-lg border border-slate-800/70 text-xs"
                          >
                            <div className="min-w-0 flex-1">
                              <div className="flex items-center gap-2">
                                <span className="font-bold text-sky-400">
                                  {isDecrease ? `-${log.amount}` : `+${log.amount}`} {goal.unit}
                                </span>
                                {log.note && (
                                  <span className="text-slate-300 truncate">
                                    "{log.note}"
                                  </span>
                                )}
                              </div>
                              <span className="text-[10px] text-slate-500">
                                {log.date} {log.time ? `• ${log.time}` : ''} • Qoldiq: {log.resultingRemaining} {goal.unit}
                              </span>
                            </div>

                            {!goal.isArchived && (
                              <button
                                onClick={() => {
                                  if (window.confirm("Ushbu qaydni bekor qilasizmi? Maqsaddan qaytarib olinadi.")) {
                                    deleteGoalProgressLog(goal.id, log.id);
                                  }
                                }}
                                className="text-slate-500 hover:text-rose-400 p-1.5 rounded-lg transition-colors ml-2"
                                title="Qaydni bekor qilish"
                              >
                                <Trash2 className="w-3.5 h-3.5" />
                              </button>
                            )}
                          </div>
                        ))}
                      </div>
                    )}
                  </div>
                )}
              </div>
            );
          })
        )}
      </div>

      {/* Manual Quantity Log Modal with note and preview */}
      {selectedGoalIdForLog && (() => {
        const targetGoal = goals.find(g => g.id === selectedGoalIdForLog);
        if (!targetGoal) return null;
        const isDec = targetGoal.goalType === 'decrease';

        return (
          <div className="fixed inset-0 z-50 bg-black/70 backdrop-blur-sm flex items-center justify-center p-4">
            <div className="bg-slate-900 border border-slate-800 rounded-3xl w-full max-w-sm p-5 shadow-2xl space-y-4">
              <div className="flex items-center justify-between pb-2 border-b border-slate-800">
                <div>
                  <h3 className="font-bold text-sm text-slate-100">Maqsadga aniq natija qo'shish</h3>
                  <p className="text-[11px] text-slate-400 truncate">{targetGoal.title}</p>
                </div>
                <button
                  onClick={() => setSelectedGoalIdForLog(null)}
                  className="text-slate-400 hover:text-white p-1 rounded-lg hover:bg-slate-800"
                >
                  <X className="w-4 h-4" />
                </button>
              </div>

              <form onSubmit={handleApplyLog} className="space-y-3.5">
                <div>
                  <label className="block text-xs font-medium text-slate-300 mb-1">
                    {isDec ? `Vazn kamayishi (${targetGoal.unit})` : `Bajarilgan miqdor (${targetGoal.unit})`} *
                  </label>
                  <input
                    type="number"
                    step={isDec ? '0.1' : '1'}
                    value={logAmount}
                    onChange={e => setLogAmount(Number(e.target.value))}
                    required
                    className="w-full bg-slate-800 border border-slate-700 rounded-xl px-3.5 py-2 text-sm text-slate-100 focus:outline-none focus:border-sky-500 font-bold"
                  />
                </div>

                {/* Quick Presets */}
                <div className="flex items-center gap-1.5">
                  {(isDec ? [0.5, 1, 1.5, 2] : [10, 20, 30, 50]).map(v => (
                    <button
                      key={v}
                      type="button"
                      onClick={() => setLogAmount(v)}
                      className="flex-1 py-1 rounded-lg text-xs bg-slate-800 hover:bg-slate-700 text-slate-300 border border-slate-700"
                    >
                      +{v}
                    </button>
                  ))}
                </div>

                <div>
                  <label className="block text-xs font-medium text-slate-300 mb-1">
                    Izoh / Aniq kitob yoki mashq nomi (Ixtiyoriy)
                  </label>
                  <input
                    type="text"
                    value={logNote}
                    onChange={e => setLogNote(e.target.value)}
                    placeholder={
                      isDec
                        ? "Haftalik o'lchov yoki izoh..."
                        : "Izoh yoki bajarilgan ish nomi..."
                    }
                    className="w-full bg-slate-800 border border-slate-700 rounded-xl px-3.5 py-2 text-xs text-slate-100 placeholder-slate-500 focus:outline-none focus:border-sky-500"
                  />
                  <p className="text-[10px] text-slate-400 mt-1">
                    Bu faqat ushbu maqsadga qo'shiladi. Boshqa kitob yoki odatlaringizga aralashmaydi.
                  </p>
                </div>

                {/* Live Preview of Calculation */}
                <div className="bg-sky-950/40 p-2.5 rounded-xl border border-sky-500/30 text-xs space-y-1">
                  <span className="font-semibold text-sky-300 block text-[11px]">Qoldiq hisobi:</span>
                  <div className="text-slate-200">
                    {isDec ? (
                      <>
                        Hozirgi: {targetGoal.currentValue} {targetGoal.unit} → Yangi: {(targetGoal.currentValue - logAmount).toFixed(1)} {targetGoal.unit}
                        <br />
                        <span className="text-emerald-400 font-bold">
                          Qoldiq: {Math.max(0, targetGoal.currentValue - logAmount - targetGoal.targetValue).toFixed(1)} {targetGoal.unit}
                        </span>
                      </>
                    ) : (
                      <>
                        {targetGoal.targetValue} - ({targetGoal.currentValue} + {logAmount}) ={' '}
                        <span className="text-emerald-400 font-bold">
                          {Math.max(0, targetGoal.targetValue - (targetGoal.currentValue + logAmount))} {targetGoal.unit} qoladi
                        </span>
                      </>
                    )}
                  </div>
                </div>

                <div className="pt-2 flex items-center gap-2">
                  <button
                    type="button"
                    onClick={() => setSelectedGoalIdForLog(null)}
                    className="flex-1 bg-slate-800 hover:bg-slate-700 text-slate-300 text-xs font-medium py-2 rounded-xl"
                  >
                    Bekor qilish
                  </button>
                  <button
                    type="submit"
                    className="flex-1 bg-sky-600 hover:bg-sky-500 text-white text-xs font-semibold py-2 rounded-xl shadow-md active:scale-95"
                  >
                    Saqlash
                  </button>
                </div>
              </form>
            </div>
          </div>
        );
      })()}

      {/* Add Goal Modal */}
      {isAddModalOpen && (
        <div className="fixed inset-0 z-50 bg-black/70 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-slate-900 border border-slate-800 rounded-3xl w-full max-w-md p-5 shadow-2xl space-y-4 animate-in fade-in zoom-in-95 duration-200">
            <div className="flex items-center justify-between pb-2 border-b border-slate-800">
              <h3 className="font-bold text-base text-slate-100">Yangi Maqsad Belgilash</h3>
              <button
                onClick={() => setIsAddModalOpen(false)}
                className="text-slate-400 hover:text-white p-1 rounded-lg hover:bg-slate-800"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleCreateGoal} className="space-y-3.5">
              {/* Goal Type Switcher */}
              <div>
                <label className="block text-xs font-medium text-slate-300 mb-1.5">
                  Maqsad turi *
                </label>
                <div className="grid grid-cols-2 gap-2">
                  <button
                    type="button"
                    onClick={() => {
                      setGoalType('reach');
                      setUnit('');
                      setTargetValue(100);
                    }}
                    className={`p-2.5 rounded-xl border text-left text-xs transition-all ${
                      goalType === 'reach'
                        ? 'bg-sky-600/20 border-sky-500 text-sky-200 font-semibold shadow-sm'
                        : 'bg-slate-800 border-slate-700 text-slate-400 hover:text-slate-200'
                    }`}
                  >
                    <span className="block font-bold">Natijaga erishish</span>
                    <span className="text-[10px] text-slate-400">Belgilangan songa oshib borish</span>
                  </button>

                  <button
                    type="button"
                    onClick={() => {
                      setGoalType('decrease');
                      setUnit('kg');
                      setStartValue(90);
                      setTargetValue(80);
                    }}
                    className={`p-2.5 rounded-xl border text-left text-xs transition-all ${
                      goalType === 'decrease'
                        ? 'bg-rose-600/20 border-rose-500 text-rose-200 font-semibold shadow-sm'
                        : 'bg-slate-800 border-slate-700 text-slate-400 hover:text-slate-200'
                    }`}
                  >
                    <span className="block font-bold">Kamaytirish / Tushirish</span>
                    <span className="text-[10px] text-slate-400">Ko'rsatkichni kamaytirib borish</span>
                  </button>
                </div>
              </div>

              <div>
                <label className="block text-xs font-medium text-slate-300 mb-1">
                  Maqsad nomi *
                </label>
                <input
                  type="text"
                  value={title}
                  onChange={e => setTitle(e.target.value)}
                  placeholder="Maqsad nomini kiriting..."
                  required
                  className="w-full bg-slate-800 border border-slate-700 rounded-xl px-3.5 py-2.5 text-sm text-slate-100 placeholder-slate-500 focus:outline-none focus:border-sky-500"
                />
              </div>

              {goalType === 'decrease' ? (
                <div className="grid grid-cols-3 gap-2">
                  <div>
                    <label className="block text-xs font-medium text-slate-300 mb-1">
                      Boshlang'ich
                    </label>
                    <input
                      type="number"
                      step="0.5"
                      value={startValue}
                      onChange={e => setStartValue(Number(e.target.value))}
                      required
                      className="w-full bg-slate-800 border border-slate-700 rounded-xl px-2.5 py-2 text-xs text-slate-100 focus:outline-none focus:border-sky-500 font-bold"
                    />
                  </div>

    setViewDate(today);
    setSelectedDate(formatDateYMD(today));
  };

  for (let i = 0; i < adjustedFirstDay; i++) {
    daysGrid.push(null);
  }

    const dStr = `${currentYear}-${String(currentMonth + 1).padStart(2, '0')}-${String(d).padStart(2, '0')}`;

  "time": string | null,
  "quantity": number | null,
  "unit": string | null,
  "explanationUz": string,
  "confidence": number
}
`;

    const generatePromise = ai.models.generateContent({
      model: 'gemini-3.8-flash',
      contents: prompt,
      config: {
        responseMimeType: 'application/json',
      },
    });

    const timeoutPromise = new Promise<never>((_, reject) =>

    const parsedJson = JSON.parse(response.text || '{}');

      source: 'gemini-ai',
    });
  } catch (error: any) {

    res.json({
      success: false,
      fallbackToLocal: true,
      message: 'Lokal qoidalar ishlatiladi',
    });
  }
});

async function startServer() {
  const isProd = process.env.NODE_ENV === 'production';

  if (!isProd) {
    const { createServer: createViteServer } = await import('vite');
    const vite = await createViteServer({
      server: { middlewareMode: true },
      appType: 'spa',
    });
    app.use(vite.middlewares);
  } else {
    app.use(express.static(path.resolve(__dirname, 'dist')));
    app.get('*', (_req: Request, res: Response) => {

    });
  }

  app.listen(PORT, '0.0.0.0', () => {

}
